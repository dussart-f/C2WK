#Config ADM

# Autoriser le trafic local et les flux déjà établis (réponses)
sudo iptables -A INPUT -i lo -j ACCEPT
sudo iptables -A INPUT -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT

# Autoriser SSH (port 22) depuis le réseau Développement (192.168.10.0/24)
sudo iptables -A INPUT -p tcp --dport 22 -s 192.168.10.0/24 -j ACCEPT

# Autoriser Samba/Partage fichiers (port 445) depuis les Services Communs (192.168.50.0/24)
sudo iptables -A INPUT -p tcp --dport 445 -s 192.168.50.0/24 -j ACCEPT

sudo iptables -A INPUT -j DROP # Bloquer tout le reste
