# config myrouter.etna.nx
# Activer le routage IP
sudo nano /etc/sysctl.conf
net.ipv4.ip_forward = 1


# Configuration des interfaces
sudo nano /etc/network/interfaces

# IN FILE
# The loopback network interface
auto lo
iface lo inet loopback

# DHCP config for ens4
auto ens4
iface ens4 inet dhcp

# Static config for ens5
auto ens5
iface ens5 inet static
        address 192.168.10.1
        netmask 255.255.255.0
        vlan-raw-device ens5

# Static config for ens6
auto ens6
iface ens6 inet static
        address 192.168.20.1
        netmask 255.255.255.0
        vlan-raw-device ens6

# Static config for ens7
auto ens7
iface ens7 inet static
        address 192.168.30.1
        netmask 255.255.255.0
        vlan-raw-device ens7

# Static config for ens8
auto ens8
iface ens8 inet static
        address 192.168.40.1
        netmask 255.255.255.0
        vlan-raw-device ens8

# Static config for ens9
auto ens9
iface ens9 inet static
        address 192.168.50.1
        netmask 255.255.255.0
        vlan-raw-device ens9

sudo systemctl restart networking

# Activer le NAT pour l'accès à Internet
sudo apt update
sudo apt install iptables-persistent
sudo iptables -t nat -A POSTROUTING -o ens4 -j MASQUERADE
sudo netfilter-persistent save

# changement du nom host
sudo nano /etc/hostname
sudo nano /etc/hosts
